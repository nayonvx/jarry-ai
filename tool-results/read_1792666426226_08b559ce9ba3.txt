    58→    58→function hexToRgb(hex: string): [number, number, number] {
    59→    59→  const h = hex.replace("#", "");
    60→    60→  return [
    61→    61→    parseInt(h.slice(0, 2), 16),
    62→    62→    parseInt(h.slice(2, 4), 16),
    63→    63→    parseInt(h.slice(4, 6), 16),
    64→    64→  ];
    65→    65→}

    67→    67→function rgba(hex: string, a: number): string {
    68→    68→  const [r, g, b] = hexToRgb(hex);
    69→    69→  return `rgba(${r}, ${g}, ${b}, ${a})`;
    70→    70→}
    71→    71→
    72→    72→function mixHex(hex: string, other: string, ratio: number): string {
    73→    73→  const [r1, g1, b1] = hexToRgb(hex);
    74→    74→  const [r2, g2, b2] = hexToRgb(other);
    75→    75→  const m = (x: number, y: number) => Math.round(x * (1 - ratio) + y * ratio);
    76→    76→  const to = (n: number) => n.toString(16).padStart(2, "0");
    77→    77→  return `#${to(m(r1, r2))}${to(m(g1, g2))}${to(m(b1, b2))}`;
    78→    78→}
    79→    79→
    80→    80→function clampHex(value: unknown, fallback: string): string {
    81→    81→  if (typeof value !== "string") return fallback;
    82→    82→  let s = value.trim().replace(/^#/, "").toLowerCase();
    83→    83→  if (/^[0-9a-f]{3}$/.test(s)) s = s.split("").map((c) => c + c).join("");
    84→    84→  if (!/^[0-9a-f]{6}$/.test(s)) return fallback;
    85→    85→  return `#${s}`;
    86→    86→}
