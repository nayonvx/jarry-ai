    58→function hexToRgb(hex: string): [number, number, number] {
    59→  const h = hex.replace("#", "");
    60→  return [
    61→    parseInt(h.slice(0, 2), 16),
    62→    parseInt(h.slice(2, 4), 16),
    63→    parseInt(h.slice(4, 6), 16),
    64→  ];
    65→}
    66→
    67→function rgba(hex: string, a: number): string {
    68→  const [r, g, b] = hexToRgb(hex);
    69→  return `rgba(${r}, ${g}, ${b}, ${a})`;
    70→}
    71→
    72→function mixHex(hex: string, other: string, ratio: number): string {
    73→  const [r1, g1, b1] = hexToRgb(hex);
    74→  const [r2, g2, b2] = hexToRgb(other);
    75→  const m = (x: number, y: number) => Math.round(x * (1 - ratio) + y * ratio);
    76→  const to = (n: number) => n.toString(16).padStart(2, "0");
    77→  return `#${to(m(r1, r2))}${to(m(g1, g2))}${to(m(b1, b2))}`;
    78→}
    79→
    80→function clampHex(value: unknown, fallback: string): string {
    81→  if (typeof value !== "string") return fallback;
    82→  let s = value.trim().replace(/^#/, "").toLowerCase();
    83→  if (/^[0-9a-f]{3}$/.test(s)) s = s.split("").map((c) => c + c).join("");
    84→  if (!/^[0-9a-f]{6}$/.test(s)) return fallback;
    85→  return `#${s}`;
    86→}

    88→function sanitizeTheme(raw: unknown): Theme {
    89→  const obj = (raw && typeof raw === "object" ? raw : {}) as Record<string, unknown>;
    90→  const name =
    91→    typeof obj.name === "string" && obj.name.trim()
    92→      ? obj.name.trim().slice(0, 40)
    93→      : "নতুন থিম";
    94→  return {
    95→    name,
    96→    bg: clampHex(obj.bg, DEFAULT_THEME.bg),
    97→    accent1: clampHex(obj.accent1, DEFAULT_THEME.accent1),
    98→    accent2: clampHex(obj.accent2, DEFAULT_THEME.accent2),
    99→  };
   100→}

   102→function buildCssVars(t: Theme): Record<string, string> {
   103→  let bg = t.bg;
   104→  const [r, g, b] = hexToRgb(bg);
   105→  // Keep the background dark enough for white text to stay readable
   106→  if (0.2126 * r + 0.7152 * g + 0.0722 * b > 96) bg = mixHex(bg, "#000000", 0.55);
   107→  const { accent1: a1, accent2: a2 } = t;
   108→  return {
   109→    "--jt-bg": bg,
   110→    "--jt-bar": rgba(bg, 0.8),
   111→    "--jt-code": mixHex(bg, "#ffffff", 0.05),
   112→    "--jt-accent1": a1,
   113→    "--jt-accent2": a2,
   114→    "--jt-accent1-a14": rgba(a1, 0.14),
   115→    "--jt-accent1-a25": rgba(a1, 0.25),
   116→    "--jt-accent2-a25": rgba(a2, 0.25),
   117→    "--jt-accent1-a40": rgba(a1, 0.45),
   118→    "--jt-accent1-tint": mixHex(a1, "#ffffff", 0.5),
   119→    "--jt-accent2-tint": mixHex(a2, "#ffffff", 0.45),
   120→    "--jt-glow-a": rgba(a1, 0.35),
   121→    "--jt-glow-b": rgba(a2, 0.5),
   122→    "--jt-glow1": rgba(a1, 0.09),
   123→    "--jt-glow2": rgba(a2, 0.08),
   124→    "--jt-glow3": rgba(a2, 0.06),
   125→  };
   126→}

   180→function CodeBlock({ code, lang }: { code: string; lang: string }) {
   181→  const [copied, setCopied] = useState(false);
   182→  const onCopy = async () => {
   183→    try {
   184→      await navigator.clipboard.writeText(code);
   185→      setCopied(true);
   186→      setTimeout(() => setCopied(false), 1500);
   187→    } catch {
   188→      /* clipboard unavailable */
   189→    }
   190→  };
   191→  return (
   192→    <div className="jt-code my-2 overflow-hidden rounded-xl border border-white/10">
   193→      <div className="flex items-center justify-between border-b border-white/10 bg-white/[0.03] px-3 py-1.5">
   194→        <span className="jt-lang font-mono text-[11px] uppercase tracking-wider">
   195→          {lang || "code"}
   196→        </span>
   197→        <button
   198→          onClick={onCopy}
   199→          className="flex items-center gap-1 rounded-md px-2 py-0.5 text-[11px] text-white/60 transition hover:bg-white/10 hover:text-white"
   200→          aria-label="কোড কপি করো"
   201→          type="button"
   202→        >
   203→          {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   204→          {copied ? "কপি হয়েছে" : "কপি"}
   205→        </button>
   206→      </div>
   207→      <SyntaxHighlighter
   208→        language={lang || "text"}
   209→        style={oneDark}
   210→        wrapLongLines
   211→        customStyle={{
   212→          margin: 0,
   213→          background: "transparent",
   214→          fontSize: "12.5px",
   215→          padding: "12px 14px",
   216→        }}
   217→        codeTagProps={{
   218→          style: {
   219→            fontFamily: "var(--font-geist-mono), ui-monospace, monospace",

   342→export default function Home() {
   343→  const [messages, setMessages] = useState<Msg[]>([]);
   344→  const [input, setInput] = useState("");
   345→  const [busy, setBusy] = useState(false);
   346→  const [webSearch, setWebSearch] = useState(false);
   347→  const [hydrated, setHydrated] = useState(false);
   348→  const [theme, setTheme] = useState<Theme>(DEFAULT_THEME);
   349→  const [themeNotice, setThemeNotice] = useState<string | null>(null);
   350→
   351→  const scrollRef = useRef<HTMLDivElement>(null);
   352→  const pinnedRef = useRef(true);
   353→  const abortRef = useRef<AbortController | null>(null);
   354→  const taRef = useRef<HTMLTextAreaElement>(null);
   355→  const noticeTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   356→
   357→  const applyCssVars = useCallback((t: Theme) => {
   358→    const root = document.documentElement;
   359→    const vars = buildCssVars(t);
   360→    for (const [k, v] of Object.entries(vars)) root.style.setProperty(k, v);
   361→  }, []);
   362→
   363→  // Load saved theme (and apply default immediately to avoid flash)
   364→  useEffect(() => {
   365→    let t = DEFAULT_THEME;
   366→    try {
   367→      const raw = localStorage.getItem(THEME_KEY);
   368→      if (raw) t = sanitizeTheme(JSON.parse(raw));
   369→    } catch {
   370→      /* ignore corrupt storage */
   371→    }
   372→    setTheme(t);
   373→    applyCssVars(t);
   374→  }, [applyCssVars]);
   375→
   376→  // Re-apply + persist on theme change
   377→  useEffect(() => {
   378→    applyCssVars(theme);
   379→    try {
   380→      localStorage.setItem(THEME_KEY, JSON.stringify(theme));
   381→    } catch {
   382→      /* storage unavailable */
   383→    }
   384→  }, [theme, applyCssVars]);
   385→
   386→  const showThemeNotice = useCallback((text: string) => {
   387→    setThemeNotice(text);
   388→    if (noticeTimer.current) clearTimeout(noticeTimer.current);
   389→    noticeTimer.current = setTimeout(() => setThemeNotice(null), 4500);
   390→  }, []);
   391→
   392→  const handleThemeEvent = useCallback(
   393→    (raw: unknown) => {
   394→      const t = sanitizeTheme(raw);
   395→      setTheme(t);
   396→      showThemeNotice(`🎨 নতুন থিম: ${t.name}`);
   397→    },
   398→    [showThemeNotice]
   399→  );
   400→
   401→  const resetTheme = () => {
   402→    setTheme(DEFAULT_THEME);
   403→    showThemeNotice("🎨 থিম রিসেট হয়েছে — Rose Night");
   404→  };
   405→
   406→  // Load saved history
   407→  useEffect(() => {
   408→    try {
   409→      const raw = localStorage.getItem(STORAGE_KEY);
   410→      if (raw) {
   411→        const parsed = JSON.parse(raw);
   412→        if (Array.isArray(parsed)) setMessages(parsed);
   413→      }
   414→    } catch {
   415→      /* ignore corrupt storage */
   416→    }
   417→    setHydrated(true);
   418→  }, []);
   419→
   420→  // Persist history
   421→  useEffect(() => {
   422→    if (!hydrated) return;
   423→    try {
   424→      localStorage.setItem(STORAGE_KEY, JSON.stringify(messages.slice(-60)));
   425→    } catch {
   426→      /* storage full/unavailable */
   427→    }
   428→  }, [messages, hydrated]);
   429→
   430→  const scrollToBottom = useCallback((force = false) => {
   431→    const el = scrollRef.current;
   432→    if (!el) return;
   433→    if (force || pinnedRef.current) el.scrollTop = el.scrollHeight;
   434→  }, []);
   435→
   436→  const onScroll = () => {
   437→    const el = scrollRef.current;
   438→    if (!el) return;
   439→    pinnedRef.current = el.scrollHeight - el.scrollTop - el.clientHeight < 90;
   440→  };
   441→
   442→  useEffect(() => {
   443→    scrollToBottom();
   444→  }, [messages, scrollToBottom]);
   445→
   446→  const autoResize = () => {
   447→    const ta = taRef.current;
   448→    if (!ta) return;
   449→    ta.style.height = "auto";
   450→    ta.style.height = `${Math.min(ta.scrollHeight, 150)}px`;
   451→  };
   452→
   453→  const send = useCallback(
   454→    async (text?: string, opts?: { search?: boolean }) => {
   455→      const content = (text ?? input).trim();
   456→      if (!content || busy) return;
   457→      const useSearch = opts?.search ?? webSearch;
   458→      if (opts?.search) setWebSearch(true);
   459→
   460→      const userMsg: Msg = { id: uid(), role: "user", content, time: Date.now() };
   461→      const aiId = uid();
   462→      const aiMsg: Msg = { id: aiId, role: "assistant", content: "", time: Date.now() };
   463→      const history = [...messages, userMsg];
   464→
   465→      setMessages([...history, aiMsg]);
   466→      setInput("");
   467→      setBusy(true);
   468→      pinnedRef.current = true;
   469→      if (taRef.current) taRef.current.style.height = "auto";
   470→
   471→      const ctrl = new AbortController();
   472→      abortRef.current = ctrl;
   473→
   474→      try {
   475→        const res = await fetch("/api/chat", {
   476→          method: "POST",
   477→          headers: { "Content-Type": "application/json" },
   478→          body: JSON.stringify({
   479→            messages: history.map((m) => ({ role: m.role, content: m.content })),
   480→            webSearch: useSearch,
   481→          }),
   482→          signal: ctrl.signal,
   483→        });
   484→
   485→        if (!res.ok || !res.body) {
   486→          const errData = await res.json().catch(() => null);
   487→          throw new Error(errData?.error || "সংযোগে সমস্যা হয়েছে!");
   488→        }
   489→
   490→        const reader = res.body.getReader();
   491→        const decoder = new TextDecoder();
   492→        let buf = "";
   493→
   494→        while (true) {
   495→          const { done, value } = await reader.read();
   496→          if (done) break;
   497→          buf += decoder.decode(value, { stream: true });
   498→          const parts = buf.split("\n\n");
   499→          buf = parts.pop() || "";
   500→
   501→          for (const part of parts) {
   502→            const line = part.trim();
   503→            if (!line.startsWith("data:")) continue;
   504→            try {
   505→              const evt = JSON.parse(line.slice(5).trim());
   506→              if (evt.type === "delta") {
   507→                setMessages((prev) =>
   508→                  prev.map((m) =>
   509→                    m.id === aiId ? { ...m, content: m.content + evt.text } : m
   510→                  )
   511→                );
   512→              } else if (evt.type === "sources") {
   513→                setMessages((prev) =>
   514→                  prev.map((m) => (m.id === aiId ? { ...m, sources: evt.sources } : m))
   515→                );
   516→              } else if (evt.type === "theme") {
   517→                handleThemeEvent(evt.theme);
   518→              } else if (evt.type === "error") {
   519→                setMessages((prev) =>
   520→                  prev.map((m) =>
   521→                    m.id === aiId
   522→                      ? { ...m, content: m.content || evt.message, error: !m.content }
   523→                      : m
   524→                  )
   525→                );
   526→              } else if (evt.type === "done") {
   527→                setMessages((prev) =>
   528→                  prev.map((m) => (m.id === aiId ? { ...m, time: m.time || Date.now() } : m))
   529→                );
   530→              }
   531→            } catch {
   532→              /* skip malformed event */
   533→            }
   534→          }
   535→        }
   536→      } catch (err: unknown) {
   537→        const e = err as Error;
   538→        if (e?.name === "AbortError") {

   540→            prev.map((m) =>
   541→              m.id === aiId
   542→                ? { ...m, content: m.content || "পাঠানো থামিয়ে দিয়েছি 😊" }

   545→          );

   548→            prev.map((m) =>
   549→              m.id === aiId

   553→                    error: !m.content,
   554→                  }
   555→                : m
   556→            )
   557→          );
   558→        }
   559→      } finally {
   560→        setBusy(false);
   561→        abortRef.current = null;
   562→      }
   563→    },
   564→    [input, busy, messages, webSearch, handleThemeEvent]
   565→  );
   566→
   567→  const stop = () => abortRef.current?.abort();
   568→

   570→    if (busy) abortRef.current?.abort();
   571→    setMessages([]);
   572→    try {
   573→      localStorage.removeItem(STORAGE_KEY);
   574→    } catch {
   575→      /* ignore */
   576→    }
   577→  };
   578→
   579→  const onKeyDown = (e: React.KeyboardEvent) => {

   581→      e.preventDefault();
   582→      send();
   583→    }
   584→  };
   585→
   586→  const isEmpty = messages.length === 0;
   587→
   588→  return (

   591→      <header className="jt-bar z-20 border-b border-white/10 backdrop-blur-xl">
   592→        <div className="mx-auto flex w-full max-w-3xl items-center justify-between gap-3 px-3 py-3 sm:px-4">
   593→          <div className="flex min-w-0 items-center gap-3">
   594→            <div className="jarry-avatar-glow shrink-0 rounded-full">
   595→              <Image
   596→                src="/jarry-avatar.png"
   597→                alt="Jarry AI avatar"
   598→                width={42}
   599→                height={42}
   600→                priority
   601→                className="jt-ring h-[42px] w-[42px] rounded-full object-cover"
   602→              />
   603→            </div>
   604→            <div className="min-w-0">
   605→              <h1 className="font-bangla-display text-xl font-bold leading-tight">
   606→                <span className="jarry-gradient-text">Jarry</span>
   607→              </h1>
   608→              <p className="flex items-center gap-1.5 text-[11px] text-white/50">
   609→                <span className="inline-block h-1.5 w-1.5 rounded-full bg-emerald-400" />
   610→                <span className="truncate">অনলাইন • তোমার বাংলা AI বন্ধু 🌸</span>
   611→              </p>
   612→            </div>
   613→          </div>
   614→
   615→          <div className="flex shrink-0 items-center gap-2">
   616→            <button
   617→              onClick={() => setWebSearch((v) => !v)}
   618→              type="button"
   619→              aria-pressed={webSearch}
   620→              title="অনলাইন সার্চ চালু/বন্ধ করো"
   621→              className={`flex h-9 items-center gap-1.5 rounded-full px-3 text-xs font-semibold transition ${
   622→                webSearch
   623→                  ? "jt-accent-bg text-white shadow-md shadow-black/30"
   624→                  : "jarry-glass text-white/70 hover:text-white"
   625→              }`}
   626→            >
   627→              <Globe className="h-3.5 w-3.5" />
   628→              <span className="hidden sm:inline">অনলাইন সার্চ</span>
   629→              <span className="sm:hidden">সার্চ</span>
   630→            </button>
   631→            <button
   632→              onClick={resetTheme}
   633→              type="button"
   634→              title="থিম রিসেট করো"
   635→              aria-label="থিম রিসেট করো"
   636→              className="jarry-glass jt-hover-accent flex h-9 w-9 items-center justify-center rounded-full text-white/60"
   637→            >
   638→              <Palette className="h-4 w-4" />
   639→            </button>
   640→            <button
   641→              onClick={clearChat}
   642→              type="button"
   643→              title="চ্যাট মুছে ফেলো"
   644→              aria-label="চ্যাট মুছে ফেলো"
   645→              className="jarry-glass jt-hover-accent flex h-9 w-9 items-center justify-center rounded-full text-white/60"
   646→            >

   648→            </button>

   651→      </header>
   652→

   657→            {themeNotice}

   669→        <div className="mx-auto flex w-full max-w-3xl flex-col gap-4">

   685→              <p className="mt-2.5 max-w-md text-[15px] leading-relaxed text-white/60">
   686→                তোমার বাংলা AI বন্ধু! কোডিং 💻, সাইবার সিকিউরিটি 🔐, অনলাইন তথ্য 🌐 —
   687→                যা জানতে চাও, বলো। বাংলাতেই কথা হবে 😊

   707→                      </span>
   708→                    </span>

   711→              </div>
   712→            </div>

   714→            messages.map((m, i) => (

   719→              />
   720→            ))
   721→          )}
   722→        </div>
   723→      </main>
   724→
   725→      {/* Input area */}

   733→                setInput(e.target.value);
   734→                autoResize();
   735→              }}
   736→              onKeyDown={onKeyDown}
   737→              rows={1}
   738→              placeholder="জ্যারিকে কিছু লিখো... (বাংলা / বাংলিশ / English)"
   739→              aria-label="মেসেজ লিখো"
   740→              className="jarry-scroll max-h-[150px] min-h-[44px] flex-1 resize-none bg-transparent px-3 py-2.5 text-[15px] text-white placeholder:text-white/35 focus:outline-none"

   762→          </div>

   764→            জ্যারি AI দ্বারা 💕 — মাঝে মাঝে ভুল হতে পারে, গুরুত্বপূর্ণ তথ্য যাচাই করে নিও ·
   765→            Enter = পাঠাও · Shift+Enter = নতুন লাইন
   766→          </p>
